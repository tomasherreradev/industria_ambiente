CAMBIO 1: BACKEND - VentasController::buscarClientes()
================================================================================

UBICACIÓN: app/Http/Controllers/VentasController.php

COMPORTAMIENTO ANTERIOR:
  - Si strlen(q) < 2 -> devolvía [] vacío.
  - Solo buscaba con 2+ caracteres.
  - Límite fijo de 10 resultados.

COMPORTAMIENTO NUEVO:
  - q vacío -> devuelve listado inicial (25 clientes activos, ordenados por razón social).
  - q con texto -> filtra por código, razón social o fantasía (ILIKE).
  - Límite configurable: ?limit=25 (mín 5, máx 50).

REEMPLAZAR el método completo por:

--------------------------------------------------------------------------------
    // API para buscar clientes
    public function buscarClientes(Request $request)
    {
        $termino = trim((string) $request->get('q', ''));
        $limite = max(5, min((int) $request->get('limit', 25), 50));

        try {
            // Solo buscar clientes principales (no sucursales)
            $query = Clientes::soloPrincipales()
                ->where('cli_estado', true);

            if ($termino !== '') {
                $query->where(function ($query) use ($termino) {
                    $query->where('cli_codigo', 'ILIKE', "%{$termino}%")
                          ->orWhere('cli_razonsocial', 'ILIKE', "%{$termino}%")
                          ->orWhere('cli_fantasia', 'ILIKE', "%{$termino}%");
                });
            } else {
                $query->orderBy('cli_razonsocial');
            }

            $clientes = $query
                ->limit($limite)
                ->get()
                ->map(function($cliente) {
                    return [
                        'id' => trim($cliente->cli_codigo),
                        'codigo' => trim($cliente->cli_codigo),
                        'text' => trim($cliente->cli_codigo) . ' - ' . trim($cliente->cli_razonsocial),
                        'razon_social' => trim($cliente->cli_razonsocial),
                        'fantasia' => $cliente->cli_fantasia ? trim($cliente->cli_fantasia) : null,
                        'direccion' => $cliente->cli_direccion ? trim($cliente->cli_direccion) : null,
                        'localidad' => $cliente->cli_localidad ? trim($cliente->cli_localidad) : null,
                        'cuit' => $cliente->cli_cuit,
                        'codigo_postal' => $cliente->cli_codigopostal ? trim($cliente->cli_codigopostal) : null,
                        'telefono' => $cliente->cli_telefono,
                        'email' => $cliente->cli_email ? trim($cliente->cli_email) : null,
                        'contacto' => $cliente->cli_contacto ? trim($cliente->cli_contacto) : null,
                        'sector' => $cliente->cli_codigocrub ? trim($cliente->cli_codigocrub) : null,
                    ];
                });

            return response()->json($clientes);

        } catch (\Exception $e) {
            Log::error('Error buscando clientes:', ['error' => $e->getMessage()]);
            return response()->json([]);
        }
    }

    function inicializarBusquedaClientes() {
        const clienteInput = document.getElementById('cliente_codigo');
        const clienteNombre = document.getElementById('cliente_nombre');
        const btnBuscarCliente = document.getElementById('btnBuscarCliente');

        if (!clienteInput || !clienteNombre || !btnBuscarCliente) {
            return;
        }

        let searchTimeout = null;
        let ultimoTermino = '';
        const cacheResultadosClientes = {};
        const cacheKeyClientes = (termino) => termino || '__inicial__';

        clienteInput.addEventListener('input', function () {
            clearTimeout(searchTimeout);
            const termino = this.value.trim();

            mostrarAyudaBusqueda(false);

            searchTimeout = setTimeout(() => {
                const clave = cacheKeyClientes(termino);
                if (clave !== ultimoTermino) {
                    ultimoTermino = clave;
                    buscarClientes(termino, clienteNombre, {
                        cache: cacheResultadosClientes,
                        cacheKey: clave,
                        listado: termino === '',
                    });
                } else if (clave in cacheResultadosClientes) {
                    mostrarResultadosClientes(cacheResultadosClientes[clave]);
                }
            }, 300);
        });

        clienteInput.addEventListener('focus', function () {
            const termino = this.value.trim();
            buscarClientes(termino, clienteNombre, {
                cache: cacheResultadosClientes,
                cacheKey: cacheKeyClientes(termino),
                forzar: true,
                listado: true,
            });
        });

        clienteInput.addEventListener('keydown', function (event) {
            if (event.key === 'Enter') {
                event.preventDefault();
                const termino = this.value.trim();
                buscarClientes(termino, clienteNombre, {
                    forzar: true,
                    cache: cacheResultadosClientes,
                    cacheKey: cacheKeyClientes(termino),
                    listado: true,
                });
            } else if (event.key === 'Escape') {
                ocultarResultadosClientes(true);
            }
        });

        btnBuscarCliente.addEventListener('click', function () {
            clienteInput.focus();
            const termino = clienteInput.value.trim();
            buscarClientes(termino, clienteNombre, {
                forzar: true,
                cache: cacheResultadosClientes,
                cacheKey: cacheKeyClientes(termino),
                listado: true,
            });
        });

        document.addEventListener('click', function (event) {
            if (!elements.clienteBuscadorWrapper) {
                return;
            }
            if (!elements.clienteBuscadorWrapper.contains(event.target)) {
                ocultarResultadosClientes(true);
            }
        });

        mostrarAyudaBusqueda(false);
    }

CAMBIOS CLAVE vs versión anterior:
  - Se eliminó el mínimo de 2 caracteres para buscar.
  - focus / botón lupa / Enter cargan listado aunque el campo esté vacío.
  - Cache usa clave '__inicial__' cuando termino está vacío.
  - listado: true evita pisar el nombre del cliente ya seleccionado (modo edición).

4.2) REEMPLAZAR función buscarClientes() COMPLETA
----------------------------------------------------------------------------

    function buscarClientes(termino, clienteNombre, opciones = {}) {
        const esListado = Boolean(opciones.listado || termino === '');
        const claveCache = opciones.cacheKey ?? (termino || '__inicial__');
        const nombreAnterior = clienteNombre ? clienteNombre.value : '';

        if (clienteNombre && !esListado) {
            clienteNombre.value = 'Buscando...';
        }

        fetch(`/api/clientes/buscar?q=${encodeURIComponent(termino)}`)
            .then(response => response.json())
            .then(data => {
                if (opciones.cache) {
                    opciones.cache[claveCache] = Array.isArray(data) ? data : [];
                }

                if (!Array.isArray(data) || data.length === 0) {
                    if (clienteNombre) {
                        clienteNombre.value = (!esListado && termino !== '')
                            ? 'Cliente no encontrado'
                            : nombreAnterior;
                    }
                    mostrarResultadosClientes([]);
                    return;
                }

                if (clienteNombre) {
                    clienteNombre.value = esListado ? nombreAnterior : '';
                }

                if (opciones.forzar || esListado) {
                    mostrarResultadosClientes(data);
                    return;
                }

                const coincidenciaExacta = data.find(
                    cliente => cliente.codigo.trim().toLowerCase() === termino.toLowerCase()
                );

                if (coincidenciaExacta) {
                    seleccionarCliente(coincidenciaExacta.codigo);
                    ocultarResultadosClientes();
                    return;
                }

                mostrarResultadosClientes(data);
            })
            .catch(error => {
                if (clienteNombre) {
                    clienteNombre.value = esListado ? nombreAnterior : 'Error en la búsqueda';
                }

                if (window.Swal) {
                    Swal.fire({
                        icon: 'error',
                        title: 'Error de Búsqueda',
                        text: 'No se pudo realizar la búsqueda de clientes. Verifique su conexión e intente nuevamente.',
                        confirmButtonText: 'Entendido',
                    });
                }
                mostrarResultadosClientes([]);
            });
    }

CAMBIOS CLAVE vs versión anterior:
  - esListado preserva clienteNombre al abrir dropdown sin filtro.
  - forzar / esListado siempre muestran dropdown (no auto-seleccionan).
  - Auto-selección por código exacto solo al tipear (sin forzar/listado).

4.3) REEMPLAZAR función mostrarResultadosClientes() COMPLETA
----------------------------------------------------------------------------

    function mostrarResultadosClientes(clientes = null) {
        if (!elements.clienteResultados) {
            return;
        }

        const contenedor = elements.clienteResultados;
        contenedor.innerHTML = '';

        if (!Array.isArray(clientes)) {
            contenedor.classList.remove('show');
            contenedor.style.display = 'none';
            elements.clienteBuscadorWrapper?.classList.remove('is-open');
            return;
        }

        if (!clientes.length) {
            const sinResultados = document.createElement('div');
            sinResultados.className = 'cliente-resultados-vacio';
            sinResultados.textContent = 'No se encontraron clientes.';
            contenedor.appendChild(sinResultados);
        } else {
            const encabezado = document.createElement('div');
            encabezado.className = 'cliente-resultados-header';
            encabezado.textContent = 'Clientes';
            contenedor.appendChild(encabezado);

            clientes.forEach(cliente => {
                const opcion = document.createElement('button');
                opcion.type = 'button';
                opcion.className = 'cliente-resultado-item';
                opcion.dataset.codigo = cliente.codigo;
                opcion.setAttribute('role', 'option');
                const razonSocial = (cliente.razon_social || '').trim();
                const fantasia = (cliente.fantasia || '').trim();
                const titulo = razonSocial || fantasia || (cliente.text || '').replace(/^\s*\S+\s*-\s*/, '').trim();
                const subtitulo = fantasia && fantasia !== titulo ? fantasia : '';
                opcion.innerHTML = subtitulo
                    ? `<span class="cliente-resultado-nombre">${escapeHtml(titulo)}</span>
                       <span class="cliente-resultado-fantasia">${escapeHtml(subtitulo)}</span>`
                    : `<span class="cliente-resultado-nombre">${escapeHtml(titulo)}</span>`;
                opcion.addEventListener('click', () => {
                    seleccionarCliente(cliente.codigo);
                });
                contenedor.appendChild(opcion);
            });
        }

        contenedor.classList.add('show');
        contenedor.style.display = 'block';
        elements.clienteBuscadorWrapper?.classList.add('is-open');
    }
